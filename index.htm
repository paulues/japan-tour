        // 탭 전환 로직

        function switchTab(tabId) {

            // 모든 탭 컨텐츠 숨기기

            document.querySelectorAll('.tab-content').forEach(el => {

                el.classList.remove('active');

            });

            // 선택된 탭 컨텐츠 보이기

            document.getElementById(`tab-${tabId}`).classList.add('active');

 

            // 네비게이션 버튼 색상 상태 변경

            document.querySelectorAll('.nav-btn').forEach(btn => {

                btn.classList.remove('text-golf', 'text-blue-500', 'text-orange-500');

                btn.classList.add('text-gray-400');

            });

 

            const activeBtn = event.currentTarget;

            activeBtn.classList.remove('text-gray-400');

            if(tabId === 'schedule') activeBtn.classList.add('text-golf');

            if(tabId === 'budget') activeBtn.classList.add('text-blue-500');

            if(tabId === 'tips') activeBtn.classList.add('text-orange-500');

        }

 

        // 초기 로드 시 탭 상태 설정

        document.addEventListener('DOMContentLoaded', () => {

            const firstBtn = document.querySelector('.nav-btn');

            firstBtn.classList.remove('text-gray-400');

            firstBtn.classList.add('text-golf');

        });

 

        // 앱 공유 버튼 로직 (현재 URL 클립보드 복사 또는 네이티브 공유)

        function shareApp() {

            if (navigator.share) {

                navigator.share({

                    title: 'ktcs 기타큐슈 골프 일정',

                    text: '재민이가 만든 3박4일 기타큐슈 골프 원정 일정표입니다!',

                    url: window.location.href

                }).catch(console.error);

            } else {

                // 네이티브 공유 미지원 시 URL 복사

                const dummy = document.createElement('input');

                document.body.appendChild(dummy);

                dummy.value = window.location.href;

                dummy.select();

                document.execCommand('copy');

                document.body.removeChild(dummy);

               

                // 알림창 대신 메시지 박스 구현 (요구사항 반영)

                const msgBox = document.createElement('div');

                msgBox.className = 'fixed top-20 left-1/2 transform -translate-x-1/2 bg-black text-white px-4 py-2 rounded shadow-lg text-sm z-50 transition-opacity duration-300';

                msgBox.textContent = '일정표 링크가 복사되었습니다! 카톡에 붙여넣기 해주세요.';

                document.body.appendChild(msgBox);

                setTimeout(() => {

                    msgBox.style.opacity = '0';

                    setTimeout(() => document.body.removeChild(msgBox), 300);

                }, 2000);

            }

        }
